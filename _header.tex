\RequirePackage{fancyhdr}

% Reserve equal-width left/right columns sized to their widest line, so the
% centered name/role block is confined to the gap between them and can never
% run into the contact info or GitHub/LinkedIn links, regardless of how long
% \role or \email end up being.
\newlength{\sidehead}
\newlength{\sideheadtmp}
\settowidth{\sidehead}{\phone}
\settowidth{\sideheadtmp}{\city}
\ifdim\sideheadtmp>\sidehead\relax\setlength{\sidehead}{\sideheadtmp}\fi
\settowidth{\sideheadtmp}{\email}
\ifdim\sideheadtmp>\sidehead\relax\setlength{\sidehead}{\sideheadtmp}\fi
\settowidth{\sideheadtmp}{\github}
\ifdim\sideheadtmp>\sidehead\relax\setlength{\sidehead}{\sideheadtmp}\fi
\settowidth{\sideheadtmp}{\LinkedIn}
\ifdim\sideheadtmp>\sidehead\relax\setlength{\sidehead}{\sideheadtmp}\fi

\fancypagestyle{fancy}{%
\fancyhf{}
\lhead{\parbox[t]{\sidehead}{\raggedright\phone \\ % PHONE
        \city \\ % CITY
	    \href{mailto:\email}{\email}}} % EMAIL
	\chead{\parbox[t]{\dimexpr\headwidth-2\sidehead-2em\relax}{%
	    \centering {\Huge \skills \name \vspace{.25em}} \\ % Name
	    {\color{highlight} \Large{\role}}}} % Role
	\rhead{\parbox[t]{\sidehead}{\raggedleft
	    \href{https://github.com/\github}{\github} \\% Github
	    \href{https://www.linkedin.com/in/\LinkedIn}{\LinkedIn}}} % LinkedIn
\renewcommand{\headrulewidth}{1pt}
\renewcommand{\headrule}{\hbox to\headwidth{%
  \color{highlight}\leaders\hrule height \headrulewidth\hfill}}
}
\pagestyle{fancy}

\setlength{\headheight}{90pt}
\setlength{\headsep}{5pt}
